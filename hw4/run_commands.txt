ROBOT: pull first (same branch as local)
cd ~/robot_testing
git pull --ff-only

ROBOT TERMINAL 1: build and launch after pulling
cd ~/robot_testing
source /opt/ros/jazzy/setup.bash &&
source rubikpi_ros2/install/setup.bash &&
colcon build --base-paths hw3 hw4 --packages-select hw3 hw4 &&
source install/setup.bash &&
export HW4_RUN_DIR="$PWD/hw4/results/$(date +%Y%m%d_%H%M%S)" &&
mkdir -p "$HW4_RUN_DIR/ros_logs" &&
git rev-parse HEAD > "$HW4_RUN_DIR/commit.txt" &&
git status --short > "$HW4_RUN_DIR/git_status.txt" &&
export ROS_LOG_DIR="$HW4_RUN_DIR/ros_logs" &&
ros2 launch hw4 hw4.launch.py environment_file:="$PWD/hw4/config/environment.json" motion_file:="$PWD/hw4/config/motion.yaml" output_dir:="$HW4_RUN_DIR"

ROBOT TERMINAL 2: start run 1 (one mission per launch)
